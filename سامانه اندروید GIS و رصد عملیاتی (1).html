import folium
from folium.plugins import MeasureControl, LocateControl
import webbrowser
import os

# ساخت نقشه پایه متمرکز روی نوار مرزی غرب و شمال‌غرب
m = folium.Map(location=[36.35, 45.15], zoom_start=8, tiles=None)

# ۱. افزودن لایه‌های نقشه پایه (توپوگرافی، ماهواره‌ای و تاریک)
folium.TileLayer(
    tiles='https://{s}.tile.opentopomap.org/{z}/{x}/{y}.png',
    attr='OpenTopoMap',
    name='نقشه توپوگرافی نظامی (ارتفاعات)'
).add_to(m)

folium.TileLayer(
    tiles='https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}',
    attr='Esri Satellite',
    name='تصاویر ماهواره‌ای (Satellite)'
).add_to(m)

folium.TileLayer('CartoDB dark_matter', name='نقشه تاریک عملیاتی').add_to(m)

# ۲. تعریف لایه‌های GIS به تفکیک موضوع
layer_bases = folium.FeatureGroup(name='لایه مقرها و اردوگاه‌ها (قرمز)')
layer_clashes = folium.FeatureGroup(name='لایه نقاط درگیری و کمین (نارنجی)')
layer_routes = folium.FeatureGroup(name='لایه محورهای تردد و قاچاق (خطوط)')
layer_zones = folium.FeatureGroup(name='لایه پلیگون مناطق تحت رصد')

# افزودن نمونه مقرها با شعاع تهدید
bases_data = [
    {"name": "مقر پشتیبانی و لجستیک ارتفاعات قندیل", "group": "گروهک تجزیه‌طلب", "coords": [36.52, 45.02], "radius": 8000},
    {"name": "اردوگاه آموزشی مرزی سیدکان", "group": "گروهک مسلح", "coords": [36.78, 44.62], "radius": 6500},
]

for b in bases_data:
    folium.Marker(
        location=b["coords"],
        popup=f"{b['name']}
گروه: {b['group']}",
        icon=folium.Icon(color='red', icon='bullseye', prefix='fa')
    ).add_to(layer_bases)
    folium.Circle(
        location=b["coords"],
        radius=b["radius"],
        color='#ef4444',
        fill=True,
        fill_opacity=0.2
    ).add_to(layer_bases)

# افزودن نمونه نقاط درگیری و رصد
clashes_data = [
    {"name": "رصد تیم مسلح در معبر کوهستانی", "date": "1405/08/18", "coords": [36.15, 45.32]},
    {"name": "کشف محموله تسلیحاتی قاچاق", "date": "1405/08/14", "coords": [35.62, 46.08]},
]

for c in clashes_data:
    folium.Marker(
        location=c["coords"],
        popup=f"{c['name']}
تاریخ: {c['date']}",
        icon=folium.Icon(color='orange', icon='exclamation-triangle', prefix='fa')
    ).add_to(layer_clashes)

# افزودن محور تردد کوهستانی (Polyline)
folium.PolyLine(
    locations=[[36.52, 45.02], [36.44, 45.18], [36.35, 45.30]],
    color='#c084fc',
    weight=4,
    dash_array='8, 8',
    popup='محور تردد و قاچاق کوهستانی شماره ۱'
).add_to(layer_routes)

# افزودن پلیگون محدوده تحت رصد
folium.Polygon(
    locations=[[36.45, 44.85], [36.65, 44.85], [36.65, 45.15], [36.45, 45.15]],
    color='#f43f5e',
    fill=True,
    fill_opacity=0.25,
    popup='محدوده تحت رصد پهپادی و اپتیکی'
).add_to(layer_zones)

# اضافه کردن تمام لایه‌ها به نقشه
layer_bases.add_to(m)
layer_clashes.add_to(m)
layer_routes.add_to(m)
layer_zones.add_to(m)

# افزودن ابزار خط‌کش، موقعیت‌یاب GPS گوشی و کنترل لایه‌ها
m.add_child(MeasureControl(primary_length_unit='kilometers'))
LocateControl(auto_start=False).add_to(m)
folium.LayerControl(collapsed=False).add_to(m)

# ذخیره و باز کردن خودکار در مرورگر اندروید
output_file = "tactical_gis_android.html"
m.save(output_file)
webbrowser.open("file://" + os.path.realpath(output_file))
print("GIS Map Successfully Launched on Android!")
